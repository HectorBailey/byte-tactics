# Multiplayer: two instances on TCP/IP, a 2-player game of up to 8 minutes (up to 11 minutes).
#
# Instance 1 hosts, instance 2 joins at 127.0.0.1. Wine's own DirectPlay TCP/IP
# provider is a stub, so the prefixes get Microsoft's (`directplay`). The game
# is Comet Catcher with cheats allowed: each side builds a Kbot lab whose
# Kbots set off for the other's base. The game ends when a commander dies
# (the guest's Kbots usually find the host's), or else the guest surrenders
# after 8 minutes; either way both see their score screen.
instances 2
directplay
reg PlayMovie dword 0
timeout 25
start
@1 every 10
@2 every 10
@1 until warning_ok 60 key Escape ; sleep 2
@2 until warning_ok 60 key Escape ; sleep 2
@1 click 377 356
@2 click 377 356
@1 waitfor mainmenu 20
@2 waitfor mainmenu 20
@1 click 187 438
@2 click 187 438
@1 waitfor select_connection 20
@2 waitfor select_connection 20

# --- the host: TCP/IP (fourth with Microsoft's DirectPlay), no address, a new game
@1 click 200 152
@1 click 532 292
@1 waitfor tcp_address 15
@1 click 487 200
@1 waitfor select_game 30
@1 click 280 352
@1 type host
@1 click 510 336
@1 waitfor create_game 15
@1 click 360 140
@1 type btgame
@1 click 510 328
@1 waitfor battle_room 30
# cheat codes allowed, the map Comet Catcher
@1 click 612 266
sleep 1
@1 click 390 315
sleep 3
@1 click 183 200
sleep 1
@1 shot map_dialog
@1 click 478 364
sleep 2

# --- the guest: TCP/IP at 127.0.0.1, join the host's game
@2 click 200 152
@2 click 532 292
@2 waitfor tcp_address 15
@2 type 127.0.0.1
@2 click 487 200
@2 waitfor select_game 30
sleep 5
@2 shot select_game
@2 click 280 352
@2 type guest
@2 click 60 134
sleep 0.5
@2 click 510 416
@2 waitfor battle_room 30
sleep 3
@2 click 488 78
sleep 3
@1 click 488 78
sleep 3
@1 shot battle_room
@2 shot battle_room
@1 click 570 453
@1 waitfor ingame 60
@2 waitfor ingame 60
sleep 3
@1 shot mp_start
@2 shot mp_start

# --- both build a Kbot lab
on skirmish_end goto game_over
@1 chat +atm
@2 chat +atm
@1 click 380 203
@2 click 380 203
sleep 1
@1 click 96 357
@2 click 96 357
sleep 0.5
@1 click 32 185
@2 click 32 185
sleep 0.5
@1 click 300 330
@2 click 300 330
sleep 1
@1 shot mp_lab_placed
@2 shot mp_lab_placed
repeat 5 @1 chat +atm ; @2 chat +atm ; sleep 9
# --- each lab builds 12 Kbots, rally point on the other base
@1 click 330 300
@2 click 330 300
sleep 1
repeat 6 @1 click 96 185 ; @2 click 96 185 ; sleep 0.3
repeat 3 @1 click 96 250 ; @2 click 96 250 ; sleep 0.3
repeat 3 @1 click 32 250 ; @2 click 32 250 ; sleep 0.3
@1 click 32 388
@2 click 32 388
sleep 0.5
@1 click 94 8
@2 click 22 114
sleep 1
@1 shot mp_queued
@2 shot mp_queued
@1 key Escape
@2 key Escape
repeat 30 @1 chat +atm ; @2 chat +atm ; sleep 9
@1 shot mp_late
@2 shot mp_late

# --- the guest surrenders
@2 key F2
@2 waitfor f2menu 5
@2 click 60 373
sleep 1.5
@2 click 384 190
sleep 1.5
@2 click 311 254
@2 waitfor mainmenu 30
@1 waitfor skirmish_end 60
label game_over
on skirmish_end off
sleep 2
@1 shot mp_end
@2 shot mp_end
@1 ifmatch victory note the host won
@1 ifmatch defeat note the host lost
@2 ifmatch victory note the guest won
@2 ifmatch defeat note the guest lost
@1 ifmatch skirmish_end click 520 424
@2 ifmatch skirmish_end click 520 424
@1 waitfor mainmenu 30
@2 waitfor mainmenu 30
@1 click 457 438
@2 click 457 438
waitexit 30
